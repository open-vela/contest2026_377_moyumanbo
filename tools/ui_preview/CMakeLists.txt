cmake_minimum_required(VERSION 3.16)
project(velasense_preview C)
get_filename_component(TEAM_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
get_filename_component(WORKSPACE_ROOT "${TEAM_ROOT}/.." ABSOLUTE)
set(LVGL_ROOT "${WORKSPACE_ROOT}/apps/graphics/lvgl/lvgl")
file(GLOB_RECURSE LVGL_SOURCES "${LVGL_ROOT}/src/*.c")
list(FILTER LVGL_SOURCES EXCLUDE REGEX "/draw/(sunxi_g2d|vg_lite|vg_lite_vector_optimize|nxp|renesas|sdl)/")
add_library(lvgl STATIC ${LVGL_SOURCES})
target_include_directories(lvgl PUBLIC "${WORKSPACE_ROOT}/apps/graphics/lvgl" "${LVGL_ROOT}" "${CMAKE_CURRENT_LIST_DIR}")
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE)
add_executable(velasense_preview preview.c "${TEAM_ROOT}/app/velasense_demo/demo_ui.c" "${TEAM_ROOT}/app/velasense_demo/fonts/lv_font_velasense_16.c")
target_include_directories(velasense_preview PRIVATE "${TEAM_ROOT}/app/velasense_demo")
target_link_libraries(velasense_preview PRIVATE lvgl m)
target_compile_options(velasense_preview PRIVATE -Wall -Wextra -Werror)
